function getPercentInfo(payload) { const explicitTier = String(payload?.rarityTier || payload?.trophyType || "") .trim() .toLowerCase(); const tierFromPayload = explicitTier === "bronze" || explicitTier === "silver" || explicitTier === "gold" || explicitTier === "sapphire" ? explicitTier : ""; const rawValue = payload?.rarityPct; const raw = rawValue === null || rawValue === undefined || rawValue === "" ? NaN : Number(rawValue); const hasPercent = Number.isFinite(raw); const value = hasPercent ? Math.max(raw, 0.1) : 50; const tier = tierFromPayload || (value > 50 ? "bronze" : value > 20 ? "silver" : value > 5 ? "gold" : "sapphire"); return { value, tier, hasPercent }; }
